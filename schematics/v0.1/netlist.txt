# 5V domain
ADAPTER2 : ESP32      # Micro-USB carries 5V + GND

# 3.3V domain
3V3    : BME.VIN , SOIL.VCC, DS18.VCC, R_DS18.1     # From the ESP32's onboard regulator
GND    : ESP32.GND , SOIL.GND , BME.GND, DS18.GND
GPIO34 : SOIL.AOUT        # ADC1, input-only, WiFi-safe
GPIO33 : BME.SDA
GPIO25 : BME.SCL
GPIO26 : DS18.DATA, R_DS18.2

# Components
SOIL      = Capacitive soil moisture sensor v1.2
BME       = BME280
ESP32     = ESP32-WROOM-32D DevKit
DS18      = DS18B20 temperature sensor
R_DS18    = 4.7 kΩ pull-up resistor
